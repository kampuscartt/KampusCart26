function money(n){return 'GH₵'+Number(n).toFixed(2)}

function showPage(page){

 document.querySelectorAll('main section').forEach(s=>s.classList.add('hidden'));

 const el=document.getElementById(page);

 if(el)el.classList.remove('hidden');

 if(page==='market')renderProducts();

function productCard(p){

 return `<article class="product">

 <img src="${escapeHTML(p.image||'https://placehold.co/500x350?text=KampusCart')}" alt="${escapeHTML(p.name)}" onerror="this.src='https://placehold.co/500x350?text=KampusCart'">

 <div class="product-info"><span class="tag">${escapeHTML(p.category)}</span>

 <h3>${escapeHTML(p.name)}</h3><p class="small">${escapeHTML(p.description||'Available on KampusCart')}</p>

 <p class="price">${money(p.price)}</p>

 <button class="btn blue" onclick="addToCart(${Number(p.id)})">Add to cart</button></div></article>`;

}

function escapeHTML(v){

 return String(v??'').replace(/[&<>"']/g,c=>({'&':'&amp;','<':'&lt;','>':'&gt;','"':'&quot;',"'":'&#39;'}[c]));

}

function renderProducts(){

 const q=(document.getElementById('search')?.value||'').toLowerCase();

 const items=allProducts().filter(p=>(activeCategory==='All'||p.category===activeCategory)&&(`${p.name} ${p.description} ${p.category}`).toLowerCase().includes(q));

 document.getElementById('products').innerHTML=items.length?items.map(productCard).join(''):'<div class="panel">No matching products found.</div>';

}

function filterCategory(c){activeCategory=c;renderProducts();showPage('market');}

function renderHome(){document.getElementById('homeProducts').innerHTML=allProducts().slice(0,4).map(productCard).join('');}

function updateCartCount(){document.getElementById('cartCount').textContent=state.cart.reduce((a,b)=>a+b.qty,0);}

function addToCart(id){

 let p=allProducts().find(x=>x.id===id);

 if(!p)return;

 let existing=state.cart.find(x=>x.id===id);

 if(existing)existing.qty++;else state.cart.push({id:p.id,qty:1});

 save();updateCartCount();alert('Added to your cart!');

}

function renderCart(){

 const box=document.getElementById('cartItems');

 if(!state.cart.length){box.innerHTML='Your cart is empty.';document.getElementById('cartTotal').textContent=money(0);return;}

 let total=0;

 box.innerHTML=state.cart.map(item=>{

 let p=allProducts().find(x=>x.id===item.id);if(!p)return '';

 total+=p.price*item.qty;

 return `<div class="cart-item"><div><b>${escapeHTML(p.name)}</b><p class="small">${money(p.price)} × ${item.qty}</p></div><button onclick="removeCart(${item.id})">Remove</button></div>`;

 }).join('');

 document.getElementById('cartTotal').textContent=money(total);

}

function removeCart(id){state.cart=state.cart.filter(x=>x.id!==id);save();updateCartCount();renderCart();}

function placeOrder(e){

 e.preventDefault();if(!state.cart.length){alert('Your cart is empty.');return;}

 let total=0;

 state.cart.forEach(i=>{let p=allProducts().find(x=>x.id===i.id);if(p)total+=p.price*i.qty;});

 const order={id:'KC'+Date.now(),name:document.getElementById('customerName').value,phone:document.getElementById('customerPhone').value,location:document.getElementById('deliveryLocation').value,address:document.getElementById('deliveryAddress').value,payment:document.getElementById('paymentMethod').value,total,status:'Order received',items:state.cart.map(i=>({...i})),date:new Date().toLocaleString()};

 state.orders.unshift(order);state.cart=[];save();updateCartCount();

 document.getElementById('checkoutMessage').innerHTML='<div class="success">Your order has been recorded in this browser. Order number: '+escapeHTML(order.id)+'</div>';

 e.target.reset();

}

function renderOrders(){

 const box=document.getElementById('ordersList');

 box.innerHTML=state.orders.length?state.orders.map(o=>`<div class="panel"><div class="row"><b>${escapeHTML(o.id)}</b><span class="badge">${escapeHTML(o.status)}</span></div><p class="small">${escapeHTML(o.date)} · ${escapeHTML(o.location)}</p><p><b>${money(o.total)}</b></p><p class="small">Payment: ${escapeHTML(o.payment)}</p></div>`).join(''):'No orders yet.';

}

function submitPrint(e){

function reportScam(e){

 e.preventDefault();state.reports.unshift({subject:reportSubject.value,details:reportDetails.value,date:new Date().toLocaleString()});save();

 document.getElementById('reportMessage').innerHTML='<div class="success">Report saved locally. In a live service it must be sent to the moderation team.</div>';e.target.reset();

}

function renderAdmin(){

 document.getElementById('adminProducts').textContent=allProducts().length;

 document.getElementById('adminOrders').textContent=state.orders.length;

 document.getElementById('adminVendors').textContent=state.vendors.length;

 document.getElementById('adminReports').textContent=state.reports.length;

 document.getElementById('adminOrderList').innerHTML=state.orders.map(o=>`<div class="cart-item"><span>${escapeHTML(o.id)} — ${escapeHTML(o.name)} — ${money(o.total)}</span><select onchange="changeStatus('${o.id}',this.value)"><option>${escapeHTML(o.status)}</option><option>Processing</option><option>Ready for dispatch</option><option>Out for delivery</option><option>Delivered</option><option>Cancelled</option></select></div>`).join('')||'No orders recorded.';

}
